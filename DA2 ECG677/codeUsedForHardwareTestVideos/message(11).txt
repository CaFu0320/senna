  /* Relative Quaternion Feature RFE - 100 Hz */

  static uint32_t last_quat_sample = 0U;
  static uint32_t last_quat_print = 0U;

  static uint8_t quat_feature_engine_started = 0U;
  static uint32_t quat_feature_sample_count = 0U;

  static float quat_feature_samples
      [QUAT_REL_TREE_WINDOW_SIZE][QUAT_REL_TREE_CHANNELS];

  uint32_t current_tick = HAL_GetTick();

  if ((current_tick - last_quat_sample) >= 10U)
  {
      MFX_input_t data_in = {0};
      MFX_output_t data_out = {0};

      uint32_t i;
      uint32_t j;
      int prediction;

      last_quat_sample = current_tick;

      /* Read IMU sensors */
      BSP_SENSOR_ACC_GetAxes(&AccValue);
      BSP_SENSOR_GYR_GetAxes(&GyrValue);
      BSP_SENSOR_MAG_GetAxes(&MagValue);

      /* Convert sensor readings to MotionFX units */
      data_in.gyro[0] = (float)GyrValue.x * FROM_MDPS_TO_DPS;
      data_in.gyro[1] = (float)GyrValue.y * FROM_MDPS_TO_DPS;
      data_in.gyro[2] = (float)GyrValue.z * FROM_MDPS_TO_DPS;

      data_in.acc[0] = (float)AccValue.x * FROM_MG_TO_G;
      data_in.acc[1] = (float)AccValue.y * FROM_MG_TO_G;
      data_in.acc[2] = (float)AccValue.z * FROM_MG_TO_G;

      data_in.mag[0] = (float)MagValue.x * FROM_MGAUSS_TO_UT50;
      data_in.mag[1] = (float)MagValue.y * FROM_MGAUSS_TO_UT50;
      data_in.mag[2] = (float)MagValue.z * FROM_MGAUSS_TO_UT50;

      /* Start MotionFX in 9-axis mode once */
      if (quat_feature_engine_started == 0U)
      {
          MotionFX_manager_start_9X();
          quat_feature_engine_started = 1U;
      }

      /* Generate quaternion at nominal 100 Hz */
      MotionFX_manager_run(&data_in, &data_out, 0.01f);

      /* Store absolute quaternion components.
         Relative conversion is performed inside the RFE classifier. */
      if (quat_feature_sample_count < QUAT_REL_TREE_WINDOW_SIZE)
      {
          for (j = 0U; j < QUAT_REL_TREE_CHANNELS; j++)
          {
              quat_feature_samples[quat_feature_sample_count][j] =
                  data_out.quaternion[j];
          }

          quat_feature_sample_count++;
      }

      /* Classify once the window is full */
      if (quat_feature_sample_count >= QUAT_REL_TREE_WINDOW_SIZE)
      {
          prediction = (int)quat_relative_feature_tree_predict_window(
              quat_feature_samples);

          /* Print prediction every 250 ms */
          if ((current_tick - last_quat_print) >= 250U)
          {
              last_quat_print = current_tick;
              UART_SendTreePrediction(prediction);
          }

          /* Shift window forward by one sample */
          for (i = 0U; i < QUAT_REL_TREE_WINDOW_SIZE - 1U; i++)
          {
              for (j = 0U; j < QUAT_REL_TREE_CHANNELS; j++)
              {
                  quat_feature_samples[i][j] =
                      quat_feature_samples[i + 1U][j];
              }
          }

          quat_feature_sample_count = QUAT_REL_TREE_WINDOW_SIZE - 1U;
      }
  }
